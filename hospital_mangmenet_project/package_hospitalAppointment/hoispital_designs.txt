    /** Generate a random alphanumeric code */
    function randomCode(length = 5) {
        return Math.random().toString(36).substring(2, length + 2).toUpperCase();
    }

    /** Generate a hospital ID like HOSP-A3X9K */
    function generateHospitalId() {
        return `HOSP-${randomCode(5)}`;
    }

    /** Generate a department ID like DEPT-CARD-003 */
    function generateDeptId(deptName, count) {
        const code = DEPARTMENT_CODES[deptName];
        if (!code) return `DEPT-UNKN-${String(count).padStart(3, '0')}`;
        return `DEPT-${code}-${String(count).padStart(3, '0')}`;
    }

    /** Generate a doctor ID like DOC-0042 */
    function generateDocId(count) {
        return `DOC-${String(count).padStart(4, '0')}`;
    }

        const container = document.getElementById('toastContainer');

        toast.innerHTML = `<i class="fas fa-${type === 'success' ? 'check-circle' : type === 'error' ? 'circle-exclamation' : 'info-circle'} mr-2"></i>${message}`;
        container.appendChild(toast);
        requestAnimationFrame(() => toast.classList.add('show'));
        setTimeout(() => {

    /** Read all hospitals from localStorage */
    function getHospitals() {
        try {
            const raw = localStorage.getItem(STORAGE_KEY);
            return raw ? JSON.parse(raw) : [];
        } catch (e) {
            console.error('Failed to parse hospital data:', e);
            return [];
        }
    }

    /** Save all hospitals to localStorage */
    function saveHospitals(data) {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    }

    /** Add a new hospital */
    function addHospital(hospital) {
        const list = getHospitals();
        list.push(hospital);
        saveHospitals(list);
    }

    /** Update an existing hospital by ID */
    function updateHospital(id, updatedData) {
        const list = getHospitals();
        const idx = list.findIndex(h => h.id === id);
        if (idx === -1) return false;
        list[idx] = updatedData;
        saveHospitals(list);
        return true;
    }

    /** Delete a hospital by ID */
    function deleteHospital(id) {
        const list = getHospitals().filter(h => h.id !== id);
        saveHospitals(list);
    }

    /** Find a hospital by ID */
    function findHospital(id) {
        return getHospitals().find(h => h.id === id) || null;
    }

    /** Read appointments */
    function getAppointments() {
        try {
            const raw = localStorage.getItem(APPOINTMENTS_KEY);
            return raw ? JSON.parse(raw) : [];
        } catch (e) {
            return [];
        }
    }

    /** Save appointments */
    function saveAppointments(data) {
        localStorage.setItem(APPOINTMENTS_KEY, JSON.stringify(data));
    }

    /** Add an appointment */
    function addAppointment(apt) {
        const list = getAppointments();
        apt.id = `APT-${randomCode(6)}`;
        apt.bookedAt = new Date().toISOString();
        list.push(apt);
        saveAppointments(list);
    }

    /** Clear all data */
    function clearAllData() {
        localStorage.removeItem(STORAGE_KEY);
        localStorage.removeItem(APPOINTMENTS_KEY);
    }

    /** Validate the admin form before saving */
    function validateAdminForm() {
        const errEl = document.getElementById('adminFormError');
        if (!adminFormState.id) { errEl.textContent = 'Please generate a Hospital ID.'; errEl.classList.remove('hidden'); return false; }
        if (!adminFormState.name.trim()) { errEl.textContent = 'Hospital name is required.'; errEl.classList.remove('hidden'); return false; }
        if (!adminFormState.location.trim()) { errEl.textContent = 'Hospital location is required.'; errEl.classList.remove('hidden'); return false; }
        if (adminFormState.departments.length === 0) { errEl.textContent = 'Add at least one department.'; errEl.classList.remove('hidden'); return false; }
        errEl.classList.add('hidden');
        return true;
    }

    /** Save the current admin form state */
    function saveAdminForm() {
        if (!validateAdminForm()) return;

        if (editingHospitalId === null) {
            addHospital(structuredClone(adminFormState));
            showToast('Hospital added successfully!');
        } else {
            updateHospital(editingHospitalId, structuredClone(adminFormState));
            showToast('Hospital updated successfully!');
        }

        resetAdminForm();
        renderAdminPreview();
        renderBookingHospitals();
    }

    /** Load a hospital into the form for editing */
    function editHospital(id) {
        const hospital = findHospital(id);
        if (!hospital) return;

        editingHospitalId = id;
        adminFormState = structuredClone(hospital);

        document.getElementById('hospitalId').value = hospital.id;
        document.getElementById('hospitalName').value = hospital.name;
        document.getElementById('hospitalLocation').value = hospital.location;
        document.getElementById('hospitalEmergency').checked = hospital.emergency;

        // Render branches
        document.getElementById('branchesList').innerHTML = '';
        hospital.branches.forEach(b => renderBranchChip(b));

        // Render departments
        document.getElementById('departmentsContainer').innerHTML = '';
        deptCounter = 0;
        hospital.departments.forEach(dept => {
            deptCounter++;
            renderDeptCard(dept);
        });

        docCounter = hospital.departments.reduce((sum, d) => sum + d.doctors.length, 0);

        // Scroll to top of form
        window.scrollTo({ top: 0, behavior: 'smooth' });
        showToast('Editing: ' + hospital.name, 'info');
    }

    function handleDeleteHospital(id) {

        if (!confirm(`Delete "${hospital.name}"? This cannot be undone.`)) return;

        deleteHospital(id);
        renderAdminPreview();
        renderBookingHospitals();
        showToast('Hospital deleted.', 'error');

        // If we were editing this hospital, reset form
        if (editingHospitalId === id) resetAdminForm();
    }

    /** Validate and submit the booking form */
    function handleBookingSubmit(e) {
        e.preventDefault();
        const errEl = document.getElementById('bookingFormError');

        const name = document.getElementById('bkName').value.trim();
        const email = document.getElementById('bkEmail').value.trim();
        const phone = document.getElementById('bkPhone').value.trim();
        const address = document.getElementById('bkAddress').value.trim();
        const hospitalId = document.getElementById('bkHospital').value;
        const department = document.getElementById('bkDepartment').value;
        const doctor = document.getElementById('bkDoctor').value;
        const timeSlot = document.getElementById('bkTime').value;
        const date = document.getElementById('bkDate').value;
        const reason = document.getElementById('bkReason').value.trim();

        // Validation
        if (!name) { showBookingError('Please enter your full name.'); return; }

        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(email)) { showBookingError('Please enter a valid email address.'); return; }

        const phoneRegex = /^[6-9]\d{9}$/;
        if (!phoneRegex.test(phone.replace(/\s/g, ''))) { showBookingError('Please enter a valid 10-digit phone number starting with 6-9.'); return; }

        if (!hospitalId) { showBookingError('Please select a hospital.'); return; }
        if (!department) { showBookingError('Please select a department.'); return; }
        if (!doctor) { showBookingError('Please select a doctor.'); return; }
        if (!timeSlot) { showBookingError('Please select a time slot.'); return; }
        if (!date) { showBookingError('Please select a date.'); return; }
        if (reason.length < 10) { showBookingError('Please provide a reason with at least 10 characters.'); return; }

        // Get hospital name
        const hospital = findHospital(hospitalId);
        const hospitalName = hospital ? hospital.name : 'Unknown';

        // Save appointment
        addAppointment({ name, email, phone, address, hospitalId, hospitalName, department, doctor, timeSlot, date, reason });

        closeModal('bookingModal');
        showToast(`Appointment booked with ${doctor} at ${hospitalName}!`);
        document.getElementById('bookingForm').reset();

        function showBookingError(msg) {
            errEl.textContent = msg;
            errEl.classList.remove('hidden');
        }
    }

    function switchPage(pageName) {
        document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
        document.querySelectorAll('.nav-tab').forEach(t => t.classList.remove('active'));

        const page = document.getElementById(`page-${pageName}`);
        const tab = document.querySelector(`.nav-tab[data-page="${pageName}"]`);

        if (page) page.classList.add('active');
        if (tab) tab.classList.add('active');

        // Re-render when switching pages
        if (pageName === 'booking') renderBookingHospitals();
        if (pageName === 'admin') renderAdminPreview();
    }

       ============================================================== */

    function bindEvents() {

        /* --- Navigation --- */
        document.querySelectorAll('.nav-tab').forEach(tab => {
            tab.addEventListener('click', () => switchPage(tab.dataset.page));
        });

        document.getElementById('navClearStorage').addEventListener('click', () => {
            if (!confirm('Clear ALL hospital and appointment data? This cannot be undone.')) return;
            clearAllData();
            resetAdminForm();
            renderAdminPreview();
            renderBookingHospitals();
            showToast('All data cleared.', 'error');
        });

        /* --- Admin: Generate ID --- */
        document.getElementById('genIdBtn').addEventListener('click', () => {
            const id = generateHospitalId();
            document.getElementById('hospitalId').value = id;
            adminFormState.id = id;
        });

        /* --- Admin: Form field changes --- */
        document.getElementById('hospitalName').addEventListener('input', function () {
            adminFormState.name = this.value;
        });

        document.getElementById('hospitalLocation').addEventListener('input', function () {
            adminFormState.location = this.value;
        });

        document.getElementById('hospitalEmergency').addEventListener('change', function () {
            adminFormState.emergency = this.checked;
        });

        /* --- Admin: Branches --- */
        document.getElementById('toggleBranchInput').addEventListener('click', () => {
            document.getElementById('branchInputArea').classList.toggle('hidden');
            document.getElementById('branchInput').focus();
        });

        document.getElementById('addBranchBtn').addEventListener('click', () => {
            const input = document.getElementById('branchInput');
            const value = input.value.trim();
            if (!value) return;
            if (adminFormState.branches.includes(value)) {
                showToast('Branch already exists.', 'error');
                return;
            }
            adminFormState.branches.push(value);
            renderBranchChip(value);
            input.value = '';
            input.focus();
        });

        document.getElementById('branchInput').addEventListener('keydown', (e) => {
            if (e.key === 'Enter') { e.preventDefault(); document.getElementById('addBranchBtn').click(); }
        });

        /* --- Admin: Departments --- */
        document.getElementById('toggleDeptInput').addEventListener('click', () => {
            const area = document.getElementById('deptInputArea');
            const isHidden = area.classList.contains('hidden');
            area.classList.toggle('hidden');
            if (isHidden) {
                document.getElementById('doctorInputArea').classList.add('hidden');
                document.getElementById('deptInput').focus();
            }
        });

        document.getElementById('saveDeptBtn').addEventListener('click', () => {
            const input = document.getElementById('deptInput');
            const deptName = input.value.trim();
            if (!deptName) { showToast('Enter a department name.', 'error'); return; }

            const exists = adminFormState.departments.find(d => d.name === deptName);
            if (exists) { showToast('Department already added.', 'error'); input.value = ''; return; }

            deptCounter++;
            const dept = {
                departmentId: generateDeptId(deptName, deptCounter),
                name: deptName,
                doctors: []
            };

            adminFormState.departments.push(dept);
            renderDeptCard(dept);
            input.value = '';
            input.focus();
            showToast(`Department "${deptName}" added.`);
        });

        document.getElementById('deptInput').addEventListener('keydown', (e) => {
            if (e.key === 'Enter') { e.preventDefault(); document.getElementById('saveDeptBtn').click(); }
        });

        /* --- Admin: Doctors --- */
        document.getElementById('saveDoctorBtn').addEventListener('click', () => {
            const input = document.getElementById('doctorInput');
            const doctorName = input.value.trim();
            if (!doctorName) { showToast('Enter a doctor name.', 'error'); return; }
            if (!activeDeptName) { showToast('No department selected.', 'error'); return; }

            const dept = adminFormState.departments.find(d => d.name === activeDeptName);
            if (!dept) { showToast('Department not found.', 'error'); return; }

            if (dept.doctors.includes(doctorName)) {
                showToast('Doctor already in this department.', 'error');
                input.value = '';
                return;
            }

            docCounter++;
            dept.doctors.push(doctorName);

            // Re-render the department card
            const card = document.querySelector(`.dept-card[data-dept-name="${activeDeptName}"]`);
            if (card) {
                const chip = document.createElement('span');
                chip.className = 'doctor-chip';
                chip.innerHTML = `
                    <i class="fas fa-user-doctor text-[10px] text-brand-400"></i>
                    ${doctorName}
                    <button type="button" class="remove-doctor" data-dept="${activeDeptName}" data-doctor="${doctorName}" title="Remove doctor">
                        <i class="fas fa-xmark text-[10px]"></i>
                    </button>
                `;
                // Remove "no doctors" text if present
                const italic = card.querySelector('.italic');
                if (italic) italic.remove();
                card.querySelector('.flex.flex-wrap').appendChild(chip);
            }

            input.value = '';
            input.focus();
            showToast(`Dr. ${doctorName} added to ${activeDeptName}.`);
        });

        document.getElementById('doctorInput').addEventListener('keydown', (e) => {
            if (e.key === 'Enter') { e.preventDefault(); document.getElementById('saveDoctorBtn').click(); }
        });

        /* --- Admin: Save & Cancel --- */
        document.getElementById('hospitalForm').addEventListener('submit', (e) => {
            e.preventDefault();
            saveAdminForm();
        });

        document.getElementById('adminCancelBtn').addEventListener('click', () => {
            if (editingHospitalId) {
                if (!confirm('Discard changes?')) return;
            }
            resetAdminForm();
            showToast('Form cleared.', 'info');
        });

        document.getElementById('adminNewHospitalBtn').addEventListener('click', () => {
            resetAdminForm();
            window.scrollTo({ top: 0, behavior: 'smooth' });
            document.getElementById('hospitalName').focus();
        });

        /* --- Admin: Delegated clicks on preview sidebar --- */
        document.getElementById('adminPreview').addEventListener('click', (e) => {
            const editBtn = e.target.closest('.edit-hospital-btn');
            const deleteBtn = e.target.closest('.delete-hospital-btn');

            if (editBtn) {
                editHospital(editBtn.dataset.id);
                return;
            }
            if (deleteBtn) {
                handleDeleteHospital(deleteBtn.dataset.id);
                return;
            }
        });

        /* --- Admin: Delegated clicks on dynamic elements --- */
        // Use document-level delegation for dynamically created elements
        document.addEventListener('click', (e) => {
            // Remove branch
            const removeBranchBtn = e.target.closest('.remove-branch');
            if (removeBranchBtn) {
                const branchName = removeBranchBtn.dataset.branch;
                adminFormState.branches = adminFormState.branches.filter(b => b !== branchName);
                removeBranchBtn.closest('.branch-chip').remove();
                return;
            }

            // Remove department
            const removeDeptBtn = e.target.closest('.remove-dept');
            if (removeDeptBtn) {
                const deptName = removeDeptBtn.dataset.dept;
                adminFormState.departments = adminFormState.departments.filter(d => d.name !== deptName);
                removeDeptBtn.closest('.dept-card').remove();
                if (activeDeptName === deptName) {
                    activeDeptName = null;
                    document.getElementById('doctorInputArea').classList.add('hidden');
                }
                return;
            }

            // Add doctor to department
            const addDoctorBtn = e.target.closest('.add-doctor-to-dept');
            if (addDoctorBtn) {
                activeDeptName = addDoctorBtn.dataset.dept;
                document.getElementById('doctorTargetDept').textContent = activeDeptName;
                document.getElementById('doctorInputArea').classList.remove('hidden');
                document.getElementById('doctorInput').focus();
                return;
            }

            // Remove doctor
            const removeDoctorBtn = e.target.closest('.remove-doctor');
            if (removeDoctorBtn) {
                const deptName = removeDoctorBtn.dataset.dept;
                const doctorName = removeDoctorBtn.dataset.doctor;
                const dept = adminFormState.departments.find(d => d.name === deptName);
                if (dept) {
                    dept.doctors = dept.doctors.filter(d => d !== doctorName);
                    removeDoctorBtn.closest('.doctor-chip').remove();

                    // If no doctors left, show placeholder
                    const card = document.querySelector(`.dept-card[data-dept-name="${deptName}"]`);
                    if (card && dept.doctors.length === 0) {
                        const wrap = card.querySelector('.flex.flex-wrap');
                        wrap.innerHTML = '<span class="text-surface-500 text-xs italic">No doctors added yet.</span>';
                    }
                }
                return;
            }

            // Booking page: Book button on hospital card
            const bookHospBtn = e.target.closest('.book-hosp-btn');
            if (bookHospBtn) {
                openBookingForHospital(bookHospBtn.dataset.hospitalId);
                return;
            }

            // Booking page: View appointments button
            const viewAptsBtn = e.target.closest('.view-apts-btn');
            if (viewAptsBtn) {
                showHospitalAppointments(viewAptsBtn.dataset.hospitalId);
                return;
            }
        });


        /* --- Booking: Hero buttons --- */
        document.getElementById('heroBookBtn').addEventListener('click', () => {
            openGenericBookingModal();
        });

        document.getElementById('heroCancelBtn').addEventListener('click', () => {
            const apts = getAppointments();
            if (apts.length === 0) {
                showToast('No appointments to cancel.', 'info');
                return;
            }
            // Show last appointment for cancellation context
            const last = apts[apts.length - 1];
            if (confirm(`Cancel your last appointment?\n\nDoctor: ${last.doctor}\nDate: ${last.date}\nTime: ${last.timeSlot}`)) {
                const list = getAppointments().filter(a => a.id !== last.id);
                saveAppointments(list);
                showToast('Appointment cancelled.', 'error');
            }
        });

        /* --- Booking: Modal --- */
        document.getElementById('closeBookingModal').addEventListener('click', () => closeModal('bookingModal'));
        document.getElementById('bookingCancelBtn').addEventListener('click', () => closeModal('bookingModal'));
        document.getElementById('bookingForm').addEventListener('submit', handleBookingSubmit);

        // Department → Doctor cascade in booking modal
        document.getElementById('bkHospital').addEventListener('change', function () {
            const hospital = findHospital(this.value);
            const deptSelect = document.getElementById('bkDepartment');
            const docSelect = document.getElementById('bkDoctor');

            deptSelect.innerHTML = '<option value="">Select Department</option>';
            docSelect.innerHTML = '<option value="">Select Doctor</option>';

            if (hospital) {
                hospital.departments.forEach(dept => {
                    deptSelect.innerHTML += `<option value="${dept.name}">${dept.name}</option>`;
                });
            }
        });

        document.getElementById('bkDepartment').addEventListener('change', function () {
            const hospitalId = document.getElementById('bkHospital').value;
            const hospital = findHospital(hospitalId);
            const docSelect = document.getElementById('bkDoctor');

            docSelect.innerHTML = '<option value="">Select Doctor</option>';

            if (hospital) {
                const dept = hospital.departments.find(d => d.name === this.value);
                if (dept) {
                    dept.doctors.forEach(doc => {
                        docSelect.innerHTML += `<option value="${doc}">${doc}</option>`;
                    });
                }
            }
        });

        /* --- Appointments Modal --- */
        document.getElementById('closeAppointmentsModal').addEventListener('click', () => closeModal('appointmentsModal'));

        /* --- Close modals on backdrop click --- */
        document.querySelectorAll('.modal-overlay').forEach(overlay => {
            overlay.addEventListener('click', (e) => {
                if (e.target === overlay) closeModal(overlay.id);
            });
        });

        /* --- Keyboard: Escape closes modals --- */
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                document.querySelectorAll('.modal-overlay.open').forEach(m => closeModal(m.id));
            }
        });
    }

    /** Open booking modal with all hospitals available */
    function openGenericBookingModal() {
        const hospitals = getHospitals();
        if (hospitals.length === 0) {
            showToast('No hospitals available. Add hospitals from Admin Panel first.', 'error');
            return;
        }

        document.getElementById('bookingForm').reset();
        document.getElementById('bookingFormError').classList.add('hidden');
        document.getElementById('bkDate').min = new Date().toISOString().split('T')[0];

        const hospSelect = document.getElementById('bkHospital');
        hospSelect.innerHTML = '<option value="">Select Hospital</option>';
        hospitals.forEach(h => {
            hospSelect.innerHTML += `<option value="${h.id}">${h.name}</option>`;
        });

        document.getElementById('bkDepartment').innerHTML = '<option value="">Select Department</option>';
        document.getElementById('bkDoctor').innerHTML = '<option value="">Select Doctor</option>';

        openModal('bookingModal');
    }

                to { opacity: 1; transform: translateY(0); }
            }
        `;
        document.head.appendChild(style);
    }

    document.addEventListener('DOMContentLoaded', init);

/* =============================================================
   SECTION 4: FLATTEN DOCTORS FROM HOSPITAL DATA
   Each doctor becomes: { key, name, hospitalId, hospitalName, departmentName, room }
   ============================================================= */
function flattenDoctors(){
    const result=[];
    getHospitals().forEach(h=>{
        (h.departments||[]).forEach(dept=>{
            (dept.doctors||[]).forEach(docName=>{
                result.push({
                    key: h.id+'::'+dept.name+'::'+docName,
                    name: docName,
                    hospitalId: h.id,
                    hospitalName: h.name,
                    departmentName: dept.name,
                    room: 'R-'+randCode(3)
                });
            });
        });
    });
    return result;
}

/** Get all unique departments across all hospitals */
function allDepartments(){
    const set=new Set();
    getHospitals().forEach(h=>(h.departments||[]).forEach(d=>set.add(d.name)));
    return [...set].sort();
}


/* =============================================================
   SECTION 5: SCHEDULER — CONFLICT DETECTION
   ============================================================= */
function checkConflict(docKey, day, shift, startTime, endTime, excludeId){
    const s1=startTime||SHIFT_META[shift].start;
    const e1=endTime||SHIFT_META[shift].end;
    for(const sch of getSchedules()){
        if(sch.doctorKey!==docKey||sch.day!==day||sch.id===excludeId) continue;
        const s2=sch.startTime||SHIFT_META[sch.shift].start;
        const e2=sch.endTime||SHIFT_META[sch.shift].end;
        if(overlaps(s1,e1,s2,e2)) return {conflict:true,existing:sch};
    }
    return {conflict:false,existing:null};
}


/* =============================================================
   SECTION 6: SCHEDULER — FILTER LOGIC
   ============================================================= */
let schedFilters={hospital:'',department:'',day:'',shift:'',search:''};
let poolSearch='';

function filterFlatDoctors(){
    let docs=flattenDoctors();
    if(schedFilters.hospital) docs=docs.filter(d=>d.hospitalId===schedFilters.hospital);
    if(schedFilters.department) docs=docs.filter(d=>d.departmentName===schedFilters.department);
    if(schedFilters.search){const q=schedFilters.search.toLowerCase();docs=docs.filter(d=>d.name.toLowerCase().includes(q)||d.departmentName.toLowerCase().includes(q)||d.hospitalName.toLowerCase().includes(q))}
    return docs;
}

function filterSchedules(){
    let list=getSchedules();
    if(schedFilters.hospital) list=list.filter(s=>s.hospitalId===schedFilters.hospital);
    if(schedFilters.department) list=list.filter(s=>s.departmentName===schedFilters.department);
    if(schedFilters.day) list=list.filter(s=>s.day===schedFilters.day);
    if(schedFilters.shift) list=list.filter(s=>s.shift===schedFilters.shift);
    if(schedFilters.search){const q=schedFilters.search.toLowerCase();list=list.filter(s=>s.doctorName.toLowerCase().includes(q)||s.departmentName.toLowerCase().includes(q))}
    return list;
}

function filterPoolDoctors(){
    let docs=filterFlatDoctors();
    if(poolSearch){const q=poolSearch.toLowerCase();docs=docs.filter(d=>d.name.toLowerCase().includes(q)||d.specialization?.toLowerCase().includes(q))}
    return docs;
}

/* =============================================================
   SECTION 10: SCHEDULER — DRAG & DROP
   ============================================================= */
let dragState={active:false,docKey:null,schedId:null,srcEl:null};

function bindDragEvents(){
    document.querySelectorAll('#poolList .draggable-card[draggable], #kanbanCols .draggable-card[draggable]').forEach(el=>{
        el.removeEventListener('dragstart',onDragStart);
        el.addEventListener('dragstart',onDragStart);
        el.removeEventListener('dragend',onDragEnd);
        el.addEventListener('dragend',onDragEnd);
    });
    document.querySelectorAll('.drop-zone').forEach(z=>{
        z.removeEventListener('dragover',onDragOver);
        z.addEventListener('dragover',onDragOver);
        z.removeEventListener('dragenter',onDragEnter);
        z.addEventListener('dragenter',onDragEnter);
        z.removeEventListener('dragleave',onDragLeave);
        z.addEventListener('dragleave',onDragLeave);
        z.removeEventListener('drop',onDrop);
        z.addEventListener('drop',onDrop);
    });
}

function onDragStart(e){
    dragState={active:true,docKey:this.dataset.docKey,schedId:this.dataset.schedId||null,srcEl:this};
    this.classList.add('drag-source');
    e.dataTransfer.effectAllowed='move';
    e.dataTransfer.setData('text/plain',this.dataset.docKey);
}
function onDragEnd(e){
    this.classList.remove('drag-source');
    document.querySelectorAll('.drop-zone').forEach(z=>z.classList.remove('drag-valid','drag-invalid'));
    dragState={active:false,docKey:null,schedId:null,srcEl:null};
}
function onDragOver(e){e.preventDefault();e.dataTransfer.dropEffect='move'}
function onDragEnter(e){
    e.preventDefault();
    const z=e.currentTarget;
    if(!dragState.docKey)return;
    const{conflict}=checkConflict(dragState.docKey,z.dataset.day,z.dataset.shift,SHIFT_META[z.dataset.shift].start,SHIFT_META[z.dataset.shift].end,dragState.schedId);
    z.classList.remove('drag-valid','drag-invalid');
    z.classList.add(conflict?'drag-invalid':'drag-valid');
}
function onDragLeave(e){if(!e.currentTarget.contains(e.relatedTarget))e.currentTarget.classList.remove('drag-valid','drag-invalid')}

function onDrop(e){
    e.preventDefault();
    const z=e.currentTarget;
    z.classList.remove('drag-valid','drag-invalid');
    const docKey=e.dataTransfer.getData('text/plain')||dragState.docKey;
    if(!docKey)return;
    const day=z.dataset.day, shift=z.dataset.shift, m=SHIFT_META[shift];

    // If moving existing schedule
    if(dragState.schedId){
        const sch=getSchedules().find(s=>s.id===dragState.schedId);
        if(sch){
            const{conflict,existing}=checkConflict(docKey,day,shift,m.start,m.end,dragState.schedId);
            if(conflict){const doc=flattenDoctors().find(d=>d.key===docKey);showToast(`Conflict: ${doc?doc.name:'Doctor'} already scheduled on ${DAY_LABELS[day]} ${SHIFT_META[existing.shift].label} (${existing.startTime||SHIFT_META[existing.shift].start}–${existing.endTime||SHIFT_META[existing.shift].end}).`,'error',5000);return}
            updateSchedule(dragState.schedId,{...sch,day,shift,startTime:m.start,endTime:m.end});
            const doc=flattenDoctors().find(d=>d.key===docKey);
            showToast(`Moved ${doc?doc.name:'doctor'} to ${DAY_LABELS[day]} ${m.label}`);
            renderScheduler();return;
        }
    }

    // New assignment
    const{conflict,existing}=checkConflict(docKey,day,shift,m.start,m.end);
    if(conflict){
        const doc=flattenDoctors().find(d=>d.key===docKey);
        showToast(`Conflict: ${doc?doc.name:'Doctor'} is already on ${DAY_LABELS[day]} ${SHIFT_META[existing.shift].label} (${existing.startTime||SHIFT_META[existing.shift].start}–${existing.endTime||SHIFT_META[existing.shift].end}). Assignment prevented.`,'error',6000);
        return;
    }
    const doc=flattenDoctors().find(d=>d.key===docKey);
    addSchedule({id:uid(),doctorKey:docKey,doctorName:doc?doc.name:'Unknown',hospitalId:doc?doc.hospitalId:'',hospitalName:doc?doc.hospitalName:'',departmentName:doc?doc.departmentName:'',day,shift,startTime:m.start,endTime:m.end,room:doc?doc.room:'',patientCount:0});
    showToast(`${doc?doc.name:'Doctor'} assigned to ${DAY_LABELS[day]} ${m.label}`);
    renderScheduler();
}

function saveScheduleFromModal(docKey,schedId){
    const day=document.getElementById('smDay').value;
    const shift=document.getElementById('smShift').value;
    const startTime=document.getElementById('smStart').value;
    const endTime=document.getElementById('smEnd').value;
    const room=document.getElementById('smRoom').value.trim();
    if(!day||!shift||!startTime||!endTime){showToast('Fill all required fields.','warning');return}
    if(toMin(endTime)<=toMin(startTime)){showToast('End must be after start time.','warning');return}
    const{conflict,existing}=checkConflict(docKey,day,shift,startTime,endTime,schedId);
    if(conflict){const cm=SHIFT_META[existing.shift];showToast(`Conflict: Already on ${DAY_LABELS[day]} ${cm.label} (${existing.startTime||cm.start}–${existing.endTime||cm.end}).`,'error',5000);return}
    const doc=flattenDoctors().find(d=>d.key===docKey);
    const data={doctorKey:docKey,doctorName:doc?doc.name:'Unknown',hospitalId:doc?doc.hospitalId:'',hospitalName:doc?doc.hospitalName:'',departmentName:doc?doc.departmentName:'',day,shift,startTime,endTime,room};
    if(schedId){updateSchedule(schedId,{...getSchedules().find(s=>s.id===schedId),...data});showToast('Schedule updated.')}
    else{addSchedule({...data,id:uid(),patientCount:0});showToast(`${doc?doc.name:'Doctor'} assigned to ${DAY_LABELS[day]} ${SHIFT_META[shift].label}`)}
    closeModal('scheduleModal');
    renderScheduler();
}

    document.getElementById('saveDeptBtn').addEventListener('click',()=>{const v=document.getElementById('deptInput').value.trim();if(!v){showToast('Enter dept name.','error');return}if(adminState.departments.find(d=>d.name===v)){showToast('Dept exists.','error');document.getElementById('deptInput').value='';return}deptCtr++;const dept={departmentId:genDeptId(v,deptCtr),name:v,doctors:[]};adminState.departments.push(dept);renderDeptCard(dept);document.getElementById('deptInput').value='';document.getElementById('deptInput').focus();showToast(`"${v}" added.`)});
    document.getElementById('deptInput').addEventListener('keydown',e=>{if(e.key==='Enter'){e.preventDefault();document.getElementById('saveDeptBtn').click()}});

    /* Admin: Doctors */
    document.getElementById('saveDocBtn').addEventListener('click',()=>{const v=document.getElementById('docInput').value.trim();if(!v){showToast('Enter doctor name.','error');return}if(!activeDept){showToast('Select department first.','error');return}const dept=adminState.departments.find(d=>d.name===activeDept);if(!dept){showToast('Dept not found.','error');return}if(dept.doctors.includes(v)){showToast('Doctor exists in dept.','error');document.getElementById('docInput').value='';return}docCtr++;dept.doctors.push(v);const card=document.querySelector(`.dept-card[data-dept-name="${activeDept}"]`);if(card){const it=card.querySelector('.italic');if(it)it.remove();const chip=document.createElement('span');chip.className='doctor-chip';chip.innerHTML=`<i class="fas fa-user-doctor text-[9px] text-brand-400"></i>${v}<button type="button" class="remove-doctor" data-dept="${activeDept}" data-doctor="${v}"><i class="fas fa-xmark text-[9px]"></i></button>`;card.querySelector('.flex.flex-wrap').appendChild(chip)}document.getElementById('docInput').value='';document.getElementById('docInput').focus();showToast(`Dr. ${v} added to ${activeDept}.`)});
    document.getElementById('docInput').addEventListener('keydown',e=>{if(e.key==='Enter'){e.preventDefault();document.getElementById('saveDocBtn').click()}});

    /* Admin: Save/Cancel/New */
    document.getElementById('hospitalForm').addEventListener('submit',e=>{e.preventDefault();saveAdminForm()});
    document.getElementById('adminCancelBtn').addEventListener('click',()=>{if(editingId&&!confirm('Discard changes?'))return;resetAdminForm();showToast('Form cleared.','info')});
    document.getElementById('adminNewBtn').addEventListener('click',()=>{resetAdminForm();window.scrollTo({top:0,behavior:'smooth'});document.getElementById('hospitalName').focus()});

    /* Admin: Preview delegation */
    document.getElementById('adminPreview').addEventListener('click',e=>{
        const eb=e.target.closest('.edit-hosp');if(eb){editHospital(eb.dataset.id);return}
        const db=e.target.closest('.del-hosp');if(db){const h=findHospital(db.dataset.id);if(!h||!confirm(`Delete "${h.name}"?`))return;deleteHospital(db.dataset.id);renderAdminPreview();renderBookingHospitals();populateSchedulerFilters();showToast('Hospital deleted.','error');if(editingId===db.dataset.id)resetAdminForm();return}
    });

    /* Booking: Hero buttons */
    document.getElementById('heroBookBtn').addEventListener('click',()=>{const hs=getHospitals();if(!hs.length){showToast('No hospitals. Add from Admin first.','error');return}document.getElementById('bookingForm').reset();document.getElementById('bkErr').classList.add('hidden');document.getElementById('bkDate').min=new Date().toISOString().split('T')[0];const sel=document.getElementById('bkHospital');sel.innerHTML='<option value="">Select Hospital</option>';hs.forEach(h=>{sel.innerHTML+=`<option value="${h.id}">${h.name}</option>`});document.getElementById('bkDepartment').innerHTML='<option value="">Select</option>';document.getElementById('bkDoctor').innerHTML='<option value="">Select</option>';openModal('bookingModal')});
    document.getElementById('heroCancelBtn').addEventListener('click',()=>{const apts=getAppointments();if(!apts.length){showToast('No appointments to cancel.','info');return}const last=apts[apts.length-1];if(confirm(`Cancel last appointment?\n\nDoctor: ${last.doctor}\nDate: ${last.date}\nTime: ${last.timeSlot||last.time||''}`)){saveAppointments(apts.filter(a=>a.id!==last.id));showToast('Appointment cancelled.','error')}});

    /* Booking: Modal */
    document.getElementById('bookingForm').addEventListener('submit',handleBookingSubmit);
    document.getElementById('bkHospital').addEventListener('change',function(){const h=findHospital(this.value);const ds=document.getElementById('bkDepartment');ds.innerHTML='<option value="">Select</option>';document.getElementById('bkDoctor').innerHTML='<option value="">Select</option>';if(h)h.departments.forEach(d=>{ds.innerHTML+=`<option value="${d.name}">${d.name}</option>`})});
    document.getElementById('bkDepartment').addEventListener('change',function(){const h=findHospital(document.getElementById('bkHospital').value);const ds=document.getElementById('bkDoctor');ds.innerHTML='<option value="">Select</option>';if(h){const dept=h.departments.find(d=>d.name===this.value);if(dept)dept.doctors.forEach(doc=>{ds.innerHTML+=`<option value="${doc}">${doc}</option>`})}});

    /* Scheduler: Filters */
    document.getElementById('schedFilterHosp').addEventListener('change',function(){schedFilters.hospital=this.value;renderScheduler()});
    document.getElementById('schedFilterDept').addEventListener('change',function(){schedFilters.department=this.value;renderScheduler()});
    document.getElementById('schedFilterDay').addEventListener('change',function(){schedFilters.day=this.value.toLowerCase();renderScheduler()});
    document.getElementById('schedFilterShift').addEventListener('change',function(){schedFilters.shift=this.value.toLowerCase();renderScheduler()});
    document.getElementById('schedSearch').addEventListener('input',debounce(function(){schedFilters.search=this.value;renderScheduler()},200));
    document.getElementById('poolSearch').addEventListener('input',debounce(function(){poolSearch=this.value;renderPool()},150));
    document.getElementById('schedClearFilters').addEventListener('click',()=>{schedFilters={hospital:'',department:'',day:'',shift:'',search:''};poolSearch='';document.getElementById('schedFilterHosp').value='';document.getElementById('schedFilterDept').value='';document.getElementById('schedFilterDay').value='';document.getElementById('schedFilterShift').value='';document.getElementById('schedSearch').value='';document.getElementById('poolSearch').value='';renderScheduler();showToast('Filters reset.','info')});

    /* Scheduler: Week nav */
    // (Week nav buttons would go here if we add them to the scheduler header)

    /* Drawer close */
    document.getElementById('drawerClose').addEventListener('click',closeDrawer);
    document.getElementById('drawerBackdrop').addEventListener('click',closeDrawer);

    /* Modal close delegation */
    document.addEventListener('click',e=>{
        if(e.target.closest('.js-close-modal'))closeModal(e.target.closest('.js-close-modal').dataset.modal);
        if(e.target.classList.contains('modal-overlay')&&!e.target.closest('.modal-box')){e.target.classList.remove('open')}
    });

    /* Global delegated clicks for dynamic elements */
    document.addEventListener('click',e=>{
        // Remove branch
        const rb=e.target.closest('.remove-branch');if(rb){adminState.branches=adminState.branches.filter(b=>b!==rb.dataset.branch);rb.closest('.branch-chip').remove();return}
        // Remove dept
        const rd=e.target.closest('.remove-dept');if(rd){adminState.departments=adminState.departments.filter(d=>d.name!==rd.dataset.dept);rd.closest('.dept-card').remove();if(activeDept===rd.dataset.dept){activeDept=null;document.getElementById('docArea').classList.add('hidden')}return}
        // Add doctor to dept
        const ad=e.target.closest('.add-doc-to-dept');if(ad){activeDept=ad.dataset.dept;document.getElementById('docTarget').textContent=activeDept;document.getElementById('docArea').classList.remove('hidden');document.getElementById('docInput').focus();return}
        // Remove doctor
        const rmd=e.target.closest('.remove-doctor');if(rmd){const dn=rmd.dataset.dept,dn2=rmd.dataset.doctor;const dept=adminState.departments.find(d=>d.name===dn);if(dept){dept.doctors=dept.doctors.filter(d=>d!==dn2);rmd.closest('.doctor-chip').remove();if(!dept.doctors.length){const card=document.querySelector(`.dept-card[data-dept-name="${dn}"]`);if(card)card.querySelector('.flex.flex-wrap').innerHTML='<span class="text-surface-500 text-[10px] italic">No doctors yet.</span>'}}return}
        // Booking: hospital cards
        const bh=e.target.closest('.book-hosp-btn');if(bh){openBookingForHospital(bh.dataset.hospId);return}
        const va=e.target.closest('.view-apts-hosp');if(va){showHospitalAppointments(va.dataset.hospId);return}
        const sh=e.target.closest('.sched-hosp-btn');if(sh){schedFilters.hospital=sh.dataset.hospId;switchPage('scheduler');return}
        // Pool: assign / view
        const pa=e.target.closest('.js-pool-assign');if(pa){openAssignModal(pa.dataset.docKey);return}
        const pv=e.target.closest('.js-pool-view-apt');if(pv){openDrawerForDoctor(pv.dataset.docKey,null);return}
        // Board: view / edit / delete
        const bv=e.target.closest('.js-bd-view');if(bv){openDrawerForDoctor(bv.dataset.docKey,bv.dataset.day);return}
        const be=e.target.closest('.js-bd-edit');if(be){openEditModal(be.dataset.schedId);return}
        const bd=e.target.closest('.js-bd-del');if(bd){openDeleteModal(bd.dataset.schedId);return}
    });

    /* Keyboard */
    document.addEventListener('keydown',e=>{if(e.key==='Escape'){document.querySelectorAll('.modal-overlay.open').forEach(m=>m.classList.remove('open'));closeDrawer()}});
}

function debounce(fn,ms){let t;return function(...a){clearTimeout(t);t=setTimeout(()=>fn.apply(this,a),ms)}}

    bindEvents();
    renderAdminPreview();
    renderBookingHospitals();
}

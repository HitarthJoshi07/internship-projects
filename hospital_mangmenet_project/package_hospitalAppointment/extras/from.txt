// ==========================================================================
// Hospital Admin Panel — state + rendering
//
// Data model
// ----------
// hospitals = [
//   {
//     id: "h_xxx",
//     name: "City General Hospital",
//     location: "New York, NY",
//     emergency: true,
//     branches: ["Queens", "Brooklyn"],
//     departments: [
//       { name: "Cardiology", doctors: ["Dr. Rao", "Dr. Singh"] }
//     ]
//   },
//   ...
// ]
//
// While the form is open (adding OR editing), all of that lives in
// `formState`, which is just a plain JS object you can read/mutate directly
// from the console or from other code:
//
//   formState.branches.push("New Branch");
//   renderBranches();
//
// On Save, formState is copied into a hospital object and pushed into (or
// replaces an entry in) `hospitals`.
// ==========================================================================

// ---- Persistent store (all saved hospitals) -----------------------------
let hospitals = [];

// ---- Draft state (the form currently being filled out) ------------------
function emptyFormState() {
    return {
        id: "",
        name: "",
        location: "",
        emergency: false,
        branches: [],
        departments: [], // { name, doctors: [] }
    };
}

let formState = emptyFormState();

// null while adding a new hospital, otherwise the id of the hospital being edited
let editingHospitalId = null;

// index of the department currently receiving a new doctor (via #add-doctor)
let activeDepartmentIndex = null;

// ---- DOM references -------------------------------------------------------
const form = document.getElementById("hospital-form");
const hospitalIdInput = document.getElementById("hospital-id");
const hospitalNameInput = document.getElementById("hospital-name");
const hospitalLocationInput = document.getElementById("hospital-location");
const hospitalEmergencyInput = document.getElementById("hospital-emergency");
const formFeedback = document.getElementById("form-feedback");
const formTitle = document.getElementById("form-title");
const formDescription = document.getElementById("form-description");

const newHospitalBtn = document.getElementById("new-hospital-btn");
const cancelBtn = document.getElementById("cancel-hospital-btn");
const saveBtn = document.getElementById("save-hospital-btn");

const hospitalListEl = document.getElementById("hospital-list");

// Branches
const addBranchBtn = document.getElementById("add-branch-btn");
const branchTemplate = document.getElementById("branch-template");
const branchInput = document.getElementById("branch-input");
const branchesListEl = document.getElementById("branches-list");

// Departments / Doctors
const addDepartmentBtn = document.getElementById("add-department-btn");
const departmentsListEl = document.getElementById("departments-list");
const addDeptBox = document.getElementById("add-dept");
const selectDeptInput = document.getElementById("select-dept");
const addDoctorBox = document.getElementById("add-doctor");
const selectDoctorInput = document.getElementById("select-doctor");

// ---- Utilities --------------------------------------------------------
function generateId() {
    return "h_" + Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
}

function showFeedback(message, isError = true) {
    formFeedback.textContent = message || "";
    formFeedback.classList.toggle("text-rose-400", isError);
    formFeedback.classList.toggle("text-emerald-400", !isError);
}

function escapeHtml(str) {
    return String(str)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;");
}

// ---- Branches -----------------------------------------------------------
addBranchBtn.addEventListener("click", () => {
    branchTemplate.classList.remove("hidden");
    branchInput.value = "";
    branchInput.focus();
});

// The template has two buttons: [data-action="add-branch"] confirms the
// entry, [data-action="remove-branch"] cancels it. Delegate on the template.
branchTemplate.addEventListener("click", (e) => {
    const action = e.target.closest("[data-action]")?.dataset.action;
    if (!action) return;

    if (action === "add-branch") {
        const value = branchInput.value.trim();
        if (!value) {
            branchInput.focus();
            return;
        }
        formState.branches.push(value);
        renderBranches();
        branchInput.value = "";
        branchInput.focus(); // stays open so you can add several in a row
    }

    if (action === "remove-branch") {
        branchTemplate.classList.add("hidden");
        branchInput.value = "";
    }
});

// Allow Enter key inside the branch input to confirm too
branchInput.addEventListener("keydown", (e) => {
    if (e.key === "Enter") {
        e.preventDefault();
        branchTemplate.querySelector('[data-action="add-branch"]').click();
    }
});

function renderBranches() {
    if (formState.branches.length === 0) {
        branchesListEl.innerHTML =
            '<p class="text-sm text-gray-400">No branches added yet.</p>';
        return;
    }

    branchesListEl.innerHTML = formState.branches
        .map(
            (branch, index) => `
        <div class="flex items-center justify-between gap-3 rounded-xl border border-gray-600 bg-gray-800 px-4 py-2 text-white">
            <span class="text-sm">${escapeHtml(branch)}</span>
            <button type="button" data-remove-branch="${index}"
                class="text-xs font-semibold text-rose-300 hover:text-rose-200">
                Remove
            </button>
        </div>`
        )
        .join("");
}

branchesListEl.addEventListener("click", (e) => {
    const btn = e.target.closest("[data-remove-branch]");
    if (!btn) return;
    const index = Number(btn.dataset.removeBranch);
    formState.branches.splice(index, 1);
    renderBranches();
});

// ---- Departments & Doctors ------------------------------------------------
addDepartmentBtn.addEventListener("click", () => {
    addDeptBox.classList.remove("hidden");
    selectDeptInput.value = "";
    selectDeptInput.focus();
});

selectDeptInput.addEventListener("keydown", (e) => {
    if (e.key !== "Enter") return;
    e.preventDefault();

    const name = selectDeptInput.value.trim();
    if (!name) return;

    const alreadyExists = formState.departments.some(
        (d) => d.name.toLowerCase() === name.toLowerCase()
    );
    if (alreadyExists) {
        showFeedback(`"${name}" is already added.`);
        return;
    }

    formState.departments.push({ name, doctors: [] });
    renderDepartments();
    showFeedback("");

    selectDeptInput.value = "";
    addDeptBox.classList.add("hidden");
});

selectDoctorInput.addEventListener("keydown", (e) => {
    if (e.key !== "Enter") return;
    e.preventDefault();

    const name = selectDoctorInput.value.trim();
    if (!name || activeDepartmentIndex === null) return;

    formState.departments[activeDepartmentIndex].doctors.push(name);
    renderDepartments();

    selectDoctorInput.value = "";
    addDoctorBox.classList.add("hidden");
    activeDepartmentIndex = null;
});

function renderDepartments() {
    // Wipe everything except the two reusable input boxes, then rebuild.
    departmentsListEl.innerHTML = "";

    if (formState.departments.length === 0) {
        const emptyMsg = document.createElement("p");
        emptyMsg.className = "text-sm text-gray-400";
        emptyMsg.textContent = "No departments added yet.";
        departmentsListEl.appendChild(emptyMsg);
    }

    formState.departments.forEach((dept, deptIndex) => {
        const card = document.createElement("div");
        card.className = "rounded-xl border border-gray-600 p-4 text-white space-y-3";

        const doctorsHtml = dept.doctors.length
            ? dept.doctors
                  .map(
                      (doc, docIndex) => `
                <div class="flex items-center justify-between gap-3 rounded-lg bg-gray-800 px-3 py-2">
                    <span class="text-sm">${escapeHtml(doc)}</span>
                    <button type="button" data-remove-doctor="${deptIndex}:${docIndex}"
                        class="text-xs font-semibold text-rose-300 hover:text-rose-200">
                        Remove
                    </button>
                </div>`
                  )
                  .join("")
            : '<p class="text-xs text-gray-400">No doctors added yet.</p>';

        card.innerHTML = `
            <div class="flex items-center justify-between gap-3">
                <h4 class="font-semibold">${escapeHtml(dept.name)}</h4>
                <div class="flex gap-2">
                    <button type="button" data-add-doctor-for="${deptIndex}"
                        class="rounded-lg bg-gray-600 px-3 py-1 text-xs font-semibold hover:bg-gray-700">
                        Add Doctor
                    </button>
                    <button type="button" data-remove-department="${deptIndex}"
                        class="rounded-lg bg-rose-600/80 px-3 py-1 text-xs font-semibold hover:bg-rose-600">
                        Remove Dept
                    </button>
                </div>
            </div>
            <div class="space-y-2">${doctorsHtml}</div>
        `;

        departmentsListEl.appendChild(card);
    });

    // Re-append the (currently hidden) reusable input boxes at the bottom
    departmentsListEl.appendChild(addDeptBox);
    departmentsListEl.appendChild(addDoctorBox);
}

departmentsListEl.addEventListener("click", (e) => {
    const removeDept = e.target.closest("[data-remove-department]");
    if (removeDept) {
        const index = Number(removeDept.dataset.removeDepartment);
        formState.departments.splice(index, 1);
        renderDepartments();
        return;
    }

    const removeDoctor = e.target.closest("[data-remove-doctor]");
    if (removeDoctor) {
        const [deptIndex, docIndex] = removeDoctor.dataset.removeDoctor
            .split(":")
            .map(Number);
        formState.departments[deptIndex].doctors.splice(docIndex, 1);
        renderDepartments();
        return;
    }

    const addDoctorFor = e.target.closest("[data-add-doctor-for]");
    if (addDoctorFor) {
        activeDepartmentIndex = Number(addDoctorFor.dataset.addDoctorFor);
        addDoctorBox.classList.remove("hidden");
        addDoctorFor.closest(".rounded-xl").appendChild(addDoctorBox);
        selectDoctorInput.value = "";
        selectDoctorInput.focus();
    }
});

// ---- Form submit: build the hospital object -----------------------------
form.addEventListener("submit", (e) => {
    e.preventDefault();

    const name = hospitalNameInput.value.trim();
    const location = hospitalLocationInput.value.trim();

    if (!name || !location) {
        showFeedback("Hospital name and location are required.");
        return;
    }

    const hospital = {
        id: hospitalIdInput.value.trim() || editingHospitalId || generateId(),
        name,
        location,
        emergency: hospitalEmergencyInput.checked,
        branches: [...formState.branches],
        departments: formState.departments.map((d) => ({
            name: d.name,
            doctors: [...d.doctors],
        })),
    };

    if (editingHospitalId) {
        const idx = hospitals.findIndex((h) => h.id === editingHospitalId);
        if (idx !== -1) hospitals[idx] = hospital;
    } else {
        hospitals.push(hospital);
    }

    showFeedback(
        editingHospitalId ? "Hospital updated." : "Hospital added.",
        false
    );

    renderHospitalList();
    resetForm();
});

// ---- Reset / cancel / new --------------------------------------------
function resetForm() {
    formState = emptyFormState();
    editingHospitalId = null;
    activeDepartmentIndex = null;

    hospitalIdInput.value = "";
    hospitalNameInput.value = "";
    hospitalLocationInput.value = "";
    hospitalEmergencyInput.checked = false;

    branchTemplate.classList.add("hidden");
    addDeptBox.classList.add("hidden");
    addDoctorBox.classList.add("hidden");

    formTitle.textContent = "Add Hospital";
    formDescription.textContent =
        "Provide hospital details and save to update the list.";
    saveBtn.textContent = "Add hospital";

    renderBranches();
    renderDepartments();
    showFeedback("");
}

newHospitalBtn.addEventListener("click", () => {
    resetForm();
    form.scrollIntoView({ behavior: "smooth", block: "start" });
});

cancelBtn.addEventListener("click", () => {
    resetForm();
});

// ---- Saved hospitals list (edit / delete) --------------------------------
function renderHospitalList() {
    if (hospitals.length === 0) {
        hospitalListEl.innerHTML =
            '<p class="text-sm text-gray-300">No hospitals added yet.</p>';
        return;
    }

    hospitalListEl.innerHTML = hospitals
        .map(
            (h) => `
        <div class="mb-3 rounded-xl border border-gray-600 bg-gray-900 p-4 text-white">
            <div class="flex items-start justify-between gap-4">
                <div>
                    <h3 class="font-semibold">${escapeHtml(h.name)}</h3>
                    <p class="text-sm text-gray-400">${escapeHtml(h.location)}</p>
                    <p class="mt-1 text-xs text-gray-400">
                        ${h.emergency ? "Emergency services available" : "No emergency services"}
                        · ${h.branches.length} branch(es)
                        · ${h.departments.length} department(s)
                    </p>
                </div>
                <div class="flex shrink-0 gap-2">
                    <button type="button" data-edit-hospital="${h.id}"
                        class="rounded-lg bg-gray-600 px-3 py-1 text-xs font-semibold hover:bg-gray-700">
                        Edit
                    </button>
                    <button type="button" data-delete-hospital="${h.id}"
                        class="rounded-lg bg-rose-600/80 px-3 py-1 text-xs font-semibold hover:bg-rose-600">
                        Delete
                    </button>
                </div>
            </div>
        </div>`
        )
        .join("");
}

hospitalListEl.addEventListener("click", (e) => {
    const editBtn = e.target.closest("[data-edit-hospital]");
    if (editBtn) {
        loadHospitalIntoForm(editBtn.dataset.editHospital);
        return;
    }

    const deleteBtn = e.target.closest("[data-delete-hospital]");
    if (deleteBtn) {
        const id = deleteBtn.dataset.deleteHospital;
        if (confirm("Delete this hospital?")) {
            hospitals = hospitals.filter((h) => h.id !== id);
            renderHospitalList();
            if (editingHospitalId === id) resetForm();
        }
    }
});

function loadHospitalIntoForm(id) {
    const hospital = hospitals.find((h) => h.id === id);
    if (!hospital) return;

    editingHospitalId = hospital.id;
    formState = {
        id: hospital.id,
        name: hospital.name,
        location: hospital.location,
        emergency: hospital.emergency,
        branches: [...hospital.branches],
        departments: hospital.departments.map((d) => ({
            name: d.name,
            doctors: [...d.doctors],
        })),
    };

    hospitalIdInput.value = hospital.id;
    hospitalNameInput.value = hospital.name;
    hospitalLocationInput.value = hospital.location;
    hospitalEmergencyInput.checked = hospital.emergency;

    formTitle.textContent = "Edit Hospital";
    formDescription.textContent = `Editing "${hospital.name}". Save to apply changes.`;
    saveBtn.textContent = "Save changes";

    renderBranches();
    renderDepartments();
    showFeedback("");

    form.scrollIntoView({ behavior: "smooth", block: "start" });
}

// ---- Init -----------------------------------------------------------------
resetForm();
renderHospitalList();
